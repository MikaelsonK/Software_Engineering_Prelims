(function() {
  const form = document.getElementById('controls');
  const countInput = document.getElementById('count');
  const nameMode = document.getElementById('nameMode');
  const results = document.getElementById('results');
  const userModal = new bootstrap.Modal(document.getElementById('userModal'));
  const modalEls = {
    picture: document.getElementById('modalPicture'),
    name: document.getElementById('modalName'),
    address: document.getElementById('modalAddress'),
    email: document.getElementById('modalEmail'),
    phone: document.getElementById('modalPhone'),
    cell: document.getElementById('modalCell'),
    dob: document.getElementById('modalDob'),
    gender: document.getElementById('modalGender')
  };
  const saveBtn = document.getElementById('saveBtn');
  const deleteBtn = document.getElementById('deleteBtn');

  let users = [];
  let currentIndex = null;

  function populate() {
    results.innerHTML = '';
    users.forEach((u, idx) => {
      const name = u && u.name ? (nameMode.value === 'first' ? u.name.first : u.name.last) : '';
      const tr = document.createElement('tr');
      tr.innerHTML = `<td>${name}</td><td>${u.gender || ''}</td><td>${u.email || ''}</td><td>${u.location?.country || ''}</td>`;
      tr.addEventListener('dblclick', () => openModal(idx));
      results.appendChild(tr);
    });
  }

  function openModal(index) {
    const u = users[index];
    currentIndex = index;
    if (!u) return;

    modalEls.picture.src = u.picture?.large || '';
    modalEls.name.value = `${u.name?.title || ''} ${u.name?.first || ''} ${u.name?.last || ''}`.trim();
    modalEls.address.value = `${u.location?.street?.number || ''} ${u.location?.street?.name || ''}, ${u.location?.city || ''}, ${u.location?.state || ''}, ${u.location?.country || ''}`;
    modalEls.email.value = u.email || '';
    modalEls.phone.value = u.phone || '';
    modalEls.cell.value = u.cell || '';
    modalEls.dob.value = u.dob?.date ? new Date(u.dob.date).toLocaleDateString() : '';
    modalEls.gender.value = u.gender || '';

    userModal.show();
  }

  saveBtn.addEventListener('click', () => {
    if (currentIndex === null) return;
    let u = users[currentIndex];
    if (!u) return;

    const [title, first, last] = modalEls.name.value.split(' ');
    u.name.title = title || u.name.title;
    u.name.first = first || u.name.first;
    u.name.last = last || u.name.last;
    u.location.street = { number: 0, name: modalEls.address.value };
    u.email = modalEls.email.value;
    u.phone = modalEls.phone.value;
    u.cell = modalEls.cell.value;
    u.dob.date = modalEls.dob.value;
    u.gender = modalEls.gender.value;

    populate();
    userModal.hide();
  });

  deleteBtn.addEventListener('click', () => {
    if (currentIndex === null) return;
    users.splice(currentIndex, 1);
    populate();
    userModal.hide();
  });

  form.addEventListener('submit', e => {
    e.preventDefault();
    const count = Number(countInput.value);
    if (!Number.isInteger(count) || count < 0 || count > 1000) return;
    if (count === 0) { users = []; results.innerHTML = ''; return; }

    fetch(`https://randomuser.me/api/?results=${count}&inc=name,gender,email,location,picture,phone,cell,dob&noinfo=true`)
      .then(response => response.json())
      .then(data => {
        users = data.results || [];
        populate();
      })
      .catch(err => console.log(err));
  });

  nameMode.addEventListener('change', () => {
    if (users.length) populate();
  });
})();
