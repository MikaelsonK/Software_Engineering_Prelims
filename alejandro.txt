    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.0.2/dist/css/bootstrap.min.css" rel="stylesheet" integrity="sha384-EVSTQN3/azprG1Anm3QDgpJLIm9Nao0Yz1ztcQTwFspd3yD65VohhpuuCOmLASjC" crossorigin="anonymous">
</head>
<body>
    <div class="container mt-5">
        <div class="d-flex mb-3 gap-2">
            <input id="resultCount" type="number" class="form-control" min="1" max="100" placeholder="Enter number of user">
            <button id="generateBtn" class="btn btn-outline-secondary">Generate</button>
        </div>
        <div id="inputError" class="alert alert-danger d-none" role="alert"></div>
        <div class="row fw-bold border-bottom pb-2 mb-2 align-items-center">
            <div class="col">
                <select id="nameType">
                    <option value="first">First Name</option>
                    <option value="last">Last Name</option>
                </select>
            </div>
            <div class="col">Gender</div>
            <div class="col">Email Address</div>
            <div class="col">Country</div>
        </div>
        <div id="results"></div>
        <div class="modal fade" id="userModal" tabindex="-1" aria-hidden="true">
            <div class="modal-dialog">
            <div class="modal-content">
            <div class="modal-header">
                <h5 class="modal-title" id="modalTitle"></h5>
                <button type="button" class="btn-close" data-bs-dismiss="modal" aria-label="Close"></button>
            </div>
            <div class="modal-body" id="modalBody"></div>
            <div class="modal-footer">
                <button id="deleteBtn" class="btn btn-danger">Delete</button>
                <button id="editBtn" class="btn btn-primary">Edit</button>
            </div>
            </div>
            </div>
        </div>
    </div>
    <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.0.2/dist/js/bootstrap.bundle.min.js" integrity="sha384-MrcW6ZMFYlzcLA8Nl+NtUVF0sA7MsXsP1UyJoMp4YLEuNSfAP+JcXn/tWtIaxVXM" crossorigin="anonymous"></script>
    <script>

    let currentUsers = [];
    let selectedIdx = null;

    function generate(users) {
        const resultsDiv = document.getElementById('results');
        resultsDiv.innerHTML = '';
        const nameType = document.getElementById('nameType').value;
        users.forEach((user, idx) => {
            const row = document.createElement('div');
            row.className = 'row mb-2 user-row';
            row.dataset.idx = idx;
            row.innerHTML = `
                <div class="col">${user.name[nameType]}</div>
                <div class="col">${user.gender}</div>
                <div class="col">${user.email}</div>
                <div class="col">${user.location.country}</div>
            `;
            row.ondblclick = () => showModal(idx);
            resultsDiv.appendChild(row);
        });
    }

    function showModal(idx) {
        selectedIdx = idx;
        const user = currentUsers[idx];
        document.getElementById('modalTitle').textContent = `${user.name.title} ${user.name.first} ${user.name.last}`;
        document.getElementById('modalBody').innerHTML = `
            <img src="${user.picture.large}" class="mb-2 rounded" style="width:100px;height:100px;object-fit:cover;">
            <div><b>Address:</b> ${user.location.street.number} ${user.location.street.name}, ${user.location.city}, ${user.location.state}, ${user.location.country}, ${user.location.postcode}</div>
            <div><b>Email:</b> ${user.email}</div>
            <div><b>Phone:</b> ${user.phone}</div>
            <div><b>Cell:</b> ${user.cell}</div>
            <div><b>DOB:</b> ${user.dob.date.slice(0,10)} (${user.dob.age} yrs)</div>
            <div><b>Gender:</b> ${user.gender}</div>
        `;
        new bootstrap.Modal(document.getElementById('userModal')).show();
    }

    document.getElementById('generateBtn').addEventListener('click', () => {
        const errorDiv = document.getElementById('inputError');
        let count = parseInt(document.getElementById('resultCount').value, 10) || 10;
        if (count < 0 || count > 1000) {
            errorDiv.textContent = 'integer must be 0-1000';
            errorDiv.classList.remove('d-none');
            return;
        } else {
            errorDiv.classList.add('d-none');
        }
        fetch(`https://randomuser.me/api/?results=${count}`)
        .then((response) => response.json())
        .then((data) => {
            currentUsers = data.results;
            generate(currentUsers);
        })
        .catch(() => console.error("Something went wrong"));
    });
    document.getElementById('nameType').addEventListener('change', function() {
        generate(currentUsers);
    });
    document.getElementById('deleteBtn').onclick = function() {
        if (selectedIdx !== null) {
            currentUsers.splice(selectedIdx, 1);
            generate(currentUsers);
            bootstrap.Modal.getInstance(document.getElementById('userModal')).hide();
        }
    };
    document.getElementById('editBtn').onclick = function() {
        if (selectedIdx !== null) {
            const user = currentUsers[selectedIdx];
            let newEmail = prompt('Edit Email:', user.email);
            if (newEmail) {
                user.email = newEmail;
                generate(currentUsers);
                document.querySelector('#modalBody div:nth-child(3)').innerHTML = `<b>Email:</b> ${user.email}`;
            }
        }
    };
    </script>
</body>
