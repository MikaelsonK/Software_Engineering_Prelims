var form = document.getElementById('getForm');
var users = [];
var currentUser = null;

form.addEventListener('submit', function(event) {
    event.preventDefault();

    var numInput = document.getElementById('numUser');
    var numUsers = numInput.value;

    fetch('https://randomuser.me/api/?results=' + numUsers)
        .then(function(response) {
            return response.json();
        })
        .then(function(data) {
            users = data.results;
            showUsers();
        })
        .catch(function(error) {
            console.log('Something went wrong' + error);
        });
});

function showUsers() {
    var tableBody = document.getElementById('userContainer');
    tableBody.innerHTML = '';
    var nameSelect = document.getElementById('nameBasis');
    var nameType = nameSelect.value;

    for (var i = 0; i < users.length; i++) {
        var user = users[i];
        var name = nameType === 'first' ? user.name.first : user.name.last;
        var row = document.createElement('tr');
        row.setAttribute('data-index', i);
        row.innerHTML = 
            '<td>' + name + '</td>' +
            '<td>' + user.gender + '</td>' +
            '<td>' + user.email + '</td>' +
            '<td>' + user.location.country + '</td>';
        row.ondblclick = function() {
            var index = this.getAttribute('data-index');
            openUserModal(index);
        };
        tableBody.appendChild(row);
    }
}

document.getElementById('nameBasis').addEventListener('change', showUsers);

function openUserModal(index) {
    currentUser = index;
    var user = users[index];
    var modalBody = document.getElementById('modalBody');
    modalBody.innerHTML = 
        '<img src="' + user.picture.large + '" class="rounded-circle mb-3" style="width:100px;height:100px;">' +
        '<div><b>' + user.name.title + ' ' + user.name.first + ' ' + user.name.last + '</b></div>' +
        '<div>' + user.gender + '</div>' +
        '<div>' + user.location.street.number + ' ' + user.location.street.name + ', ' + user.location.city + ', ' + user.location.state + ', ' + user.location.country + ', ' + user.location.postcode + '</div>' +
        '<div>Email: ' + user.email + '</div>' +
        '<div>Phone: ' + user.phone + '</div>' +
        '<div>Cell: ' + user.cell + '</div>' +
        '<div>DOB: ' + new Date(user.dob.date).toLocaleDateString() + ' (' + user.dob.age + ' years)</div>';
    var modal = new bootstrap.Modal(document.getElementById('userModal'));
    modal.show();
}

document.getElementById('deleteUserBtn').onclick = function() {
    if (currentUser !== null) {
        users.splice(currentUser, 1);
        showUsers();
        bootstrap.Modal.getInstance(document.getElementById('userModal')).hide();
    }
};

document.getElementById('editUserBtn').onclick = function() {
    if (currentUser !== null) {
        var user = users[currentUser];
        var modalBody = document.getElementById('modalBody');
        modalBody.innerHTML = 
            '<img src="' + user.picture.large + '" class="rounded-circle mb-3" style="width:100px;height:100px;">' +
            '<div><input type="text" id="editTitle" value="' + user.name.title + '"></div>' +
            '<div><input type="text" id="editFirst" value="' + user.name.first + '"></div>' +
            '<div><input type="text" id="editLast" value="' + user.name.last + '"></div>' +
            '<div><input type="text" id="editGender" value="' + user.gender + '"></div>' +
            '<div><input type="text" id="editStreetNum" value="' + user.location.street.number + '"></div>' +
            '<div><input type="text" id="editStreetName" value="' + user.location.street.name + '"></div>' +
            '<div><input type="text" id="editCity" value="' + user.location.city + '"></div>' +
            '<div><input type="text" id="editState" value="' + user.location.state + '"></div>' +
            '<div><input type="text" id="editCountry" value="' + user.location.country + '"></div>' +
            '<div><input type="text" id="editPostcode" value="' + user.location.postcode + '"></div>' +
            '<div><input type="email" id="editEmail" value="' + user.email + '"></div>' +
            '<div><input type="text" id="editPhone" value="' + user.phone + '"></div>' +
            '<div><input type="text" id="editCell" value="' + user.cell + '"></div>' +
            '<div><input type="date" id="editDob" value="' + user.dob.date.substring(0,10) + '"></div>' +
            '<button id="saveEditBtn">Save</button>';
        document.getElementById('saveEditBtn').onclick = function() {
            user.name.title = document.getElementById('editTitle').value;
            user.name.first = document.getElementById('editFirst').value;
            user.name.last = document.getElementById('editLast').value;
            user.gender = document.getElementById('editGender').value;
            user.location.street.number = document.getElementById('editStreetNum').value;
            user.location.street.name = document.getElementById('editStreetName').value;
            user.location.city = document.getElementById('editCity').value;
            user.location.state = document.getElementById('editState').value;
            user.location.country = document.getElementById('editCountry').value;
            user.location.postcode = document.getElementById('editPostcode').value;
        }
    }};
