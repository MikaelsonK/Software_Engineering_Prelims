    <script>
        document.addEventListener('DOMContentLoaded', function() {
            const userCountInput = document.getElementById('userCount');
            const nameTypeSelector = document.getElementById('nameTypeSelector');
            const usersList = document.getElementById('usersList');
            const errorMessage = document.getElementById('errorMessage');
                
            // Popup elements
            const popup = document.getElementById('popup');
            const popupUserName = document.getElementById('popupUserName');
            const popupUserAddress = document.getElementById('popupUserAddress');
            const popupUserEmail = document.getElementById('popupUserEmail');
            const popupUserImage = document.getElementById('popupUserImage');
            const closeBtn = document.querySelector('.close-btn');
                
            // Edit elements
            const editUserName = document.getElementById('editUserName');
            const editUserAddress = document.getElementById('editUserAddress');
            const editUserEmail = document.getElementById('editUserEmail');
                
            // Action buttons
            const editButton = document.getElementById('editButton');
            const deleteButton = document.getElementById('deleteButton');
            const saveButton = document.getElementById('saveButton');
            const cancelButton = document.getElementById('cancelButton');
                
            let currentUsers = [];
            let nameType = 'first';
            let timeoutId = null;
            let currentUserIndex = -1;
                
            // Change name type
            nameTypeSelector.addEventListener('change', function() {
                nameType = this.value;
                showUsers(currentUsers);
            });

            // Input event for user count
            userCountInput.addEventListener('input', function() {
                clearTimeout(timeoutId);

                timeoutId = setTimeout(function() {
                    if (userCountInput.value < 0) userCountInput.value = 0;
                    if (userCountInput.value > 1000) userCountInput.value = 1000;
                    getUsers();
                }, 1000);
            });

            // Close popup
            closeBtn.addEventListener('click', function() {
                closePopup();
            });

            // Close popup if clicked outside
            popup.addEventListener('click', function(e) {
                if (e.target === popup) {
                    closePopup();
                }
            });

            // Edit button event
            editButton.addEventListener('click', function() {
                enableEditMode();
            });

            // Delete button event
            deleteButton.addEventListener('click', function() {
                deleteUser(currentUserIndex);
                closePopup();
            });

            // Save button event
            saveButton.addEventListener('click', function() {
                saveUserChanges(currentUserIndex);
                closePopup();
            });

            // Cancel button event
            cancelButton.addEventListener('click', function() {
                disableEditMode();
                showUserDetails(currentUsers[currentUserIndex]);
            });

            function getUsers() {
                const count = parseInt(userCountInput.value);

                if (isNaN(count) || count < 0 || count > 1000) {
                    showError('Please enter a number between 0 and 1000');
                    return;
                }

                if (count === 0) {
                    usersList.innerHTML = '';
                    currentUsers = [];
                    hideError();
                    return;
                }

                hideError();

                fetch('https://randomuser.me/api/?results=' + count + '&nat=us,gb,ca,au')
                    .then(response => response.json())
                    .then(data => {
                        currentUsers = data.results;
                        showUsers(currentUsers);
                    })
                    .catch(error => {
                        showError('Error loading users. Please try again.');
                    });
            }

            function showUsers(users) {
                usersList.innerHTML = '';

                for (let i = 0; i < users.length; i++) {
                    const user = users[i];
                    const name = nameType === 'first' ? user.name.first : user.name.last;

                    const userDiv = document.createElement('div');
                    userDiv.className = 'user-row';
                    userDiv.setAttribute('data-index', i);

                    userDiv.innerHTML = `
                        <div>${name}</div>
                        <div>${user.gender.slice(0,1).toUpperCase() + user.gender.slice(1).toLowerCase()}</div>
                        <div>${user.email}</div>
                        <div>${user.location.country}</div>
                    `;

                    // Add click event to show popup
                    userDiv.addEventListener('click', function() {
                        currentUserIndex = i;
                        showPopup(user);
                    });

                    usersList.appendChild(userDiv);
                }
            }

            function showPopup(user) {
                showUserDetails(user);
                disableEditMode();
                popup.style.display = 'flex';
            }

            function showUserDetails(user) {
                const address = 
                    user.location.street.number + ' ' + 
                    user.location.street.name + ', ' + 
                    user.location.city + ', ' + 
                    user.location.state + ' ' + 
                    user.location.postcode + ', ' + 
                    user.location.country;

                popupUserName.textContent = user.name.title + ' ' + user.name.first + ' ' + user.name.last;
                popupUserAddress.textContent = address;
                popupUserEmail.textContent = user.email;
                popupUserImage.src = user.picture.large;

                // Set edit values
                editUserName.value = user.name.title + ' ' + user.name.first + ' ' + user.name.last;
                editUserAddress.value = address;
                editUserEmail.value = user.email;
            }

            function closePopup() {
                popup.style.display = 'none';
                disableEditMode();
            }

            function enableEditMode() {
                popup.querySelector('.view-mode').style.display = 'none';
                popup.querySelector('.edit-mode').style.display = 'block';
                editButton.style.display = 'none';
                deleteButton.style.display = 'none';
                saveButton.style.display = 'block';
                cancelButton.style.display = 'block';
            }

            function disableEditMode() {
                popup.querySelector('.view-mode').style.display = 'block';
                popup.querySelector('.edit-mode').style.display = 'none';
                editButton.style.display = 'block';
                deleteButton.style.display = 'block';
                saveButton.style.display = 'none';
                cancelButton.style.display = 'none';
            }

            function deleteUser(index) {
                currentUsers.splice(index, 1);
                showUsers(currentUsers);
            }

            function saveUserChanges(index) {
                const user = currentUsers[index];

                // Parse the edited name
                const editedName = editUserName.value.split(' ');
                if(editedName.length === 1){
                    user.name.first = editedName[0]
                }
                if (editedName.length >= 3) {
                    user.name.title = editedName[0];
                    user.name.first = editedName[1];
                    user.name.last = editedName.slice(2).join(' ');
                } else if (editedName.length === 2) {
                    user.name.first = editedName[0];
                    user.name.last = editedName[1];
                }

                user.email = editUserEmail.value;

                showUsers(currentUsers);
            }

            function showError(message) {
                errorMessage.textContent = message;
                errorMessage.classList.remove('d-none');
            }

            function hideError() {
                errorMessage.classList.add('d-none');
            }

            getUsers();
        });
    </script>
