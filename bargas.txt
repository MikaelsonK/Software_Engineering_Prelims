// Cache DOM references
const form = document.getElementById("fetchForm");
const amountField = document.getElementById("amount");
const nameChoice = document.getElementById("whichName");
const tableBody = document.getElementById("people");

let people = [];

// Function to render users in the table
function renderTable(data) {
  tableBody.innerHTML = "";

  data.forEach(person => {
    const row = document.createElement("tr");

    const nameCell = document.createElement("td");
    nameCell.textContent = nameChoice.value === "first" ? person.name.first : person.name.last;

    const genderCell = document.createElement("td");
    genderCell.textContent = person.gender;

    const emailCell = document.createElement("td");
    emailCell.textContent = person.email;

    const countryCell = document.createElement("td");
    countryCell.textContent = person.location.country;

    row.append(nameCell, genderCell, emailCell, countryCell);
    tableBody.appendChild(row);
  });
}

// Form submission: fetch new users
form.addEventListener("submit", async (event) => {
  event.preventDefault();
  const amt = parseInt(amountField.value, 10);

  if (isNaN(amt) || amt < 0 || amt > 1000) return;

  if (amt === 0) {
    people = [];
    tableBody.innerHTML = "";
    return;
  }

  try {
    const response = await fetch(`https://randomuser.me/api/?results=${amt}&inc=name,gender,email,location&noinfo=true`);
    const json = await response.json();
    people = json.results || [];
    renderTable(people);
  } catch (error) {
    console.error("Error fetching users:", error);
  }
});

// Update table when switching name type
nameChoice.addEventListener("change", () => {
  if (people.length > 0) {
    renderTable(people);
  }
});
